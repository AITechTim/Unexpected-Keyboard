cmake_minimum_required(VERSION 3.22)
project(prediction_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../vendor/llama.cpp llama)
add_library(cdict STATIC ../../vendor/cdict/libcdict/libcdict.c)
target_include_directories(cdict PUBLIC ../../vendor/cdict/libcdict)
add_executable(prediction-bench benchmark.cpp)
target_include_directories(prediction-bench PRIVATE ../src/main/cpp)
target_link_libraries(prediction-bench PRIVATE llama cdict)
add_executable(word-boundary-test word_boundary_test.cpp)
target_include_directories(word-boundary-test PRIVATE ../src/main/cpp)
target_link_libraries(word-boundary-test PRIVATE llama)
enable_testing()
add_test(NAME word-boundaries COMMAND word-boundary-test)
add_executable(score-server score_server.cpp)
target_include_directories(score-server PRIVATE ../src/main/cpp)
target_link_libraries(score-server PRIVATE llama)
add_executable(score-test score_test.cpp)
target_include_directories(score-test PRIVATE ../src/main/cpp)
target_link_libraries(score-test PRIVATE llama)
