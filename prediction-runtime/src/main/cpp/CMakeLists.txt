cmake_minimum_required(VERSION 3.22)
project(keyboard_prediction LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../../../vendor/llama.cpp llama)
add_library(keyboard_prediction SHARED predictor.cpp)
target_compile_features(keyboard_prediction PRIVATE cxx_std_17)
target_compile_options(keyboard_prediction PRIVATE -O3)
target_link_libraries(keyboard_prediction PRIVATE llama android log)
target_link_options(keyboard_prediction PRIVATE "-Wl,-z,max-page-size=16384")
